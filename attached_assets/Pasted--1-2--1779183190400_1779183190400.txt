```python
# 고래 신호 감지
def detect_whale_signal(trade_data):
    signals = []
    
    # 1. 단일 대량 체결 감지
    large_trades = trade_data[
        trade_data['amount'] > 500_000_000  # 5억 이상
    ]
    
    # 2. 연속 대량 체결 (같은 방향 3회 이상)
    consecutive = large_trades.groupby('direction').rolling(3).count()
    
    # 3. 전체 거래대금 대비 비율
    whale_ratio = large_trades['amount'].sum() / trade_data['amount'].sum()
    
    if whale_ratio > 0.3:  # 30% 이상이 고래 거래
        signals.append({
            'type': '긴급고래',
            'amount': large_trades['amount'].sum(),
            'count': len(large_trades)
        })
    
    return signals

# 박스권 돌파 감지
def detect_box_breakout(price_data, window=20):
    
    # 박스권 상단/하단 계산
    box_high = price_data['high'].rolling(window).max()
    box_low = price_data['low'].rolling(window).min()
    box_range = (box_high - box_low) / box_low
    
    # 횡보 구간 (변동폭 10% 이내)
    is_consolidation = box_range < 0.10
    
    # 돌파 감지
    breakout = (
        price_data['close'] > box_high.shift(1) and
        is_consolidation.rolling(5).sum() >= 4  # 최근 5일 중 4일 횡보
    )
    
    # 거래량 동반 확인
    volume_confirm = (
        price_data['volume'] > price_data['volume'].rolling(20).mean() * 2
    )
    
    return breakout & volume_confirm
```

```python
def integrated_score(quant_score, force_score, market_phase):
    
    # 시장 국면에 따라 가중치 동적 조절
    if market_phase == "상승장":
        w_quant, w_force = 0.35, 0.65  # 세력 비중 증가
    elif market_phase == "하락장":
        w_quant, w_force = 0.55, 0.45  # 퀀트(방어) 비중 증가
    else:  # 횡보장
        w_quant, w_force = 0.45, 0.55

    final = quant_score * w_quant + force_score * w_force
    
    # 등급 분류
    grade = (
        "S" if final >= 8.5 else
        "A" if final >= 7.0 else
        "B" if final >= 5.5 else
        "C"
    )
    return final, grade

# 센서뷰 적용 예시

# 수급신뢰도 0.0 → 퀀트 점수 하락 요인
```

```python
def add_quant_layer(ticker="321370"):
    
    # DART API로 재무 자동 조회
    financial = dart_api.get_financial(ticker)
    
    quant_metrics = {
        "PER": financial['market_cap'] / financial['net_income'],
        "PBR": financial['market_cap'] / financial['book_value'],
        "ROE": financial['net_income'] / financial['equity'],
        "영업이익률": financial['op_income'] / financial['revenue'],
        "부채비율": financial['debt'] / financial['equity'],
        "매출성장률": (financial['revenue'] - financial['revenue_prev'])
                      / financial['revenue_prev'] * 100
    }
    
    # 세력 분석 데이터와 결합
    force_data = {
        "세력점수": 6.4,
        "고래신호": 10.0,   # 긴급 1조 3,571억
        "박스돌파": 10.0,   # +15.3%
        "업종모멘텀": 10.0  # 시장 대비 +40.3%
    }
    
    return {**quant_metrics, **force_data}
```
